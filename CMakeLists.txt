cmake_minimum_required(VERSION 3.16)
project(assimp LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

add_library(Assimp STATIC)

# Output directories
set_target_properties(Assimp PROPERTIES
        ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/${CMAKE_BUILD_TYPE}/assimp"
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/${CMAKE_BUILD_TYPE}/assimp"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/${CMAKE_BUILD_TYPE}/assimp"
)

# Sources
file(GLOB_RECURSE ASSIMP_SRC
        "${CMAKE_CURRENT_SOURCE_DIR}/code/*.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/code/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/contrib/irrXML/*"
)
target_sources(Assimp PRIVATE ${ASSIMP_SRC})

# Include directories
target_include_directories(Assimp PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}"
        "${CMAKE_CURRENT_SOURCE_DIR}/include"
        "${CMAKE_CURRENT_SOURCE_DIR}/contrib/irrXML"
        "${CMAKE_CURRENT_SOURCE_DIR}/contrib/rapidjson/include"
)

# Platform-specific config
if(WIN32)
  target_compile_definitions(Assimp PRIVATE _WINDOWS)
endif()

# Config-specific flags
target_compile_options(Assimp PRIVATE

        $<$<CONFIG:Release>:/MT>
